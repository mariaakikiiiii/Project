CREATE OR ALTER PROCEDURE dbo.ClubLedgerPost @ClubId int,@Amount decimal(12,2),@Kind varchar(12),@RefType varchar(12),@RefId bigint,@Reference nvarchar(100)=NULL,@By int=NULL AS
BEGIN
 SET NOCOUNT ON;
 -- Caller owns the transaction. Every sale, refund and payout is posted once.
 IF EXISTS(SELECT 1 FROM dbo.ClubLedger WHERE Kind=@Kind AND RefType=@RefType AND RefId=@RefId) RETURN;
 DECLARE @Bal decimal(12,2)=(SELECT Balance FROM dbo.Clubs WITH(UPDLOCK,HOLDLOCK) WHERE ClubId=@ClubId);
 IF @Bal+@Amount<0 THROW 50004,'The club balance cannot go below zero.',1;
 UPDATE dbo.Clubs SET Balance=Balance+@Amount WHERE ClubId=@ClubId;
 INSERT dbo.ClubLedger(ClubId,Amount,BalanceAfter,Kind,RefType,RefId,Reference,CreatedBy) VALUES(@ClubId,@Amount,@Bal+@Amount,@Kind,@RefType,@RefId,@Reference,@By);
END;
GO
CREATE OR ALTER PROCEDURE dbo.WalletDebit @UserId int,@Amount decimal(12,2),@Kind varchar(20),@CourtBookingId bigint=NULL,@ClubPurchaseId bigint=NULL,@Note nvarchar(400) AS
BEGIN
 SET NOCOUNT ON;
 -- Caller owns the transaction. @Amount is signed (negative = pay, positive = refund).
 IF NOT EXISTS(SELECT 1 FROM dbo.WalletAccounts WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId) INSERT dbo.WalletAccounts(UserId) VALUES(@UserId);
 DECLARE @Bal decimal(12,2)=(SELECT Balance FROM dbo.WalletAccounts WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId);
 IF @Bal+@Amount<0 THROW 50006,'Not enough money in your GameOn wallet. Top up first.',1;
 IF @Bal+@Amount>500 THROW 50002,'A GameOn wallet can hold at most USD 500.',1;
 UPDATE dbo.WalletAccounts SET Balance=Balance+@Amount,UpdatedAt=SYSUTCDATETIME() WHERE UserId=@UserId;
 INSERT dbo.WalletLedger(UserId,Amount,BalanceAfter,Kind,CourtBookingId,ClubPurchaseId,Note) VALUES(@UserId,@Amount,@Bal+@Amount,@Kind,@CourtBookingId,@ClubPurchaseId,@Note);
END;
GO
CREATE OR ALTER PROCEDURE dbo.BookCourt @CourtId int,@UserId int,@StartsAtUtc datetime2(0),@Key uniqueidentifier,@RedemptionCode char(10)=NULL,@GuestName nvarchar(80)=NULL,@PayFromWallet bit=0 AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON; SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Club int,@Slot smallint,@Open tinyint,@Close tinyint,@Base decimal(10,2),@Peak decimal(10,2),@PS tinyint,@PE tinyint,@Active bit,@Approval bit;
  SELECT @Club=ClubId,@Slot=SlotMinutes,@Open=OpenHour,@Close=CloseHour,@Base=BasePrice,@Peak=PeakPrice,@PS=PeakStartHour,@PE=PeakEndHour,@Active=IsActive,@Approval=RequiresApproval
  FROM dbo.Courts WITH(UPDLOCK,HOLDLOCK) WHERE CourtId=@CourtId;
  IF @Club IS NULL OR @Active=0 THROW 50001,'Court not found.',1;
  DECLARE @Existing bigint=(SELECT BookingId FROM dbo.CourtBookings WHERE CreatedBy=@UserId AND IdempotencyKey=@Key);
  IF @Existing IS NOT NULL BEGIN COMMIT; SELECT BookingId,Price,ListPrice,Discount,StartsAt,EndsAt,Status,PaidFromWallet FROM dbo.CourtBookings WHERE BookingId=@Existing; RETURN; END;
  DECLARE @Staff bit=CASE WHEN @GuestName IS NOT NULL THEN 1 ELSE 0 END;
  IF @Staff=1 AND dbo.ClubRoleRank(@Club,@UserId)<2 THROW 50003,'Only club staff can book for walk-in guests.',1;
  IF @Staff=1 SET @PayFromWallet=0;
  DECLARE @Local datetime2(0)=dbo.ToBeirut(@StartsAtUtc),@End datetime2(0)=DATEADD(minute,@Slot,@StartsAtUtc);
  DECLARE @Minute int=DATEPART(hour,@Local)*60+DATEPART(minute,@Local);
  IF @StartsAtUtc<=SYSUTCDATETIME() THROW 50002,'Choose a future time.',1;
  IF @StartsAtUtc>DATEADD(day,60,SYSUTCDATETIME()) THROW 50002,'Bookings open up to 60 days ahead.',1;
  IF @Minute<@Open*60 OR @Minute+@Slot>@Close*60 OR (@Minute-@Open*60)%@Slot<>0 OR DATEPART(second,@Local)<>0 THROW 50002,'Choose one of the listed time slots.',1;
  IF EXISTS(SELECT 1 FROM dbo.CourtBlocks WHERE CourtId=@CourtId AND StartsAt<@End AND EndsAt>@StartsAtUtc) THROW 50004,'The club has blocked this time.',1;
  IF EXISTS(SELECT 1 FROM dbo.CourtBookings WHERE CourtId=@CourtId AND Status IN('Pending','Confirmed','Completed','NoShow') AND StartsAt<@End AND EndsAt>@StartsAtUtc) THROW 50004,'Someone just booked this slot. Pick another time.',1;
  IF @Staff=0 AND (SELECT COUNT(*) FROM dbo.CourtBookings WHERE UserId=@UserId AND Status IN('Pending','Confirmed') AND StartsAt>SYSUTCDATETIME())>=6 THROW 50004,'You can hold up to 6 upcoming court bookings.',1;
  DECLARE @List decimal(10,2)=CASE WHEN DATEPART(hour,@Local)>=@PS AND DATEPART(hour,@Local)<@PE THEN @Peak ELSE @Base END,@Discount decimal(10,2)=0,@ProUsed bit=0,@Redemption bigint=NULL;
  IF @Staff=0 AND dbo.IsPro(@UserId)=1
  BEGIN
   DECLARE @ProPct tinyint=(SELECT CASE WHEN IsPartner=1 THEN ProDiscountPct ELSE 0 END FROM dbo.Clubs WHERE ClubId=@Club);
   IF @ProPct>0 BEGIN SET @Discount=ROUND(@List*@ProPct/100.0,2); SET @ProUsed=1; END;
  END;
  IF @RedemptionCode IS NOT NULL
  BEGIN
   DECLARE @Pct tinyint,@RClub int;
   SELECT @Redemption=x.RedemptionId,@Pct=r.DiscountPct,@RClub=r.ClubId FROM dbo.RewardRedemptions x WITH(UPDLOCK,HOLDLOCK) JOIN dbo.Rewards r ON r.RewardId=x.RewardId
   WHERE x.Code=@RedemptionCode AND x.UserId=@UserId AND x.Status='Issued' AND x.ExpiresAt>SYSUTCDATETIME() AND r.Kind='CourtDiscount';
   IF @Redemption IS NULL THROW 50002,'That reward code is not valid, already used, or expired.',1;
   IF @RClub IS NOT NULL AND @RClub<>@Club THROW 50002,'That reward is for a different club.',1;
   SET @Discount=@Discount+ROUND((@List-@Discount)*@Pct/100.0,2);
   UPDATE dbo.RewardRedemptions SET Status='Used',UsedAt=SYSUTCDATETIME() WHERE RedemptionId=@Redemption;
  END;
  DECLARE @Status varchar(10)=CASE WHEN @Staff=0 AND @Approval=1 THEN 'Pending' ELSE 'Confirmed' END;
  INSERT dbo.CourtBookings(CourtId,UserId,GuestName,StartsAt,EndsAt,ListPrice,Discount,Status,Source,CreatedBy,IdempotencyKey,RedemptionId,ProDiscount,PaidFromWallet)
  VALUES(@CourtId,IIF(@Staff=1,NULL,@UserId),@GuestName,@StartsAtUtc,@End,@List,@Discount,@Status,IIF(@Staff=1,'Staff','Player'),@UserId,@Key,@Redemption,@ProUsed,@PayFromWallet);
  DECLARE @Id bigint=SCOPE_IDENTITY(),@Price decimal(10,2)=@List-@Discount;
  -- Paying from the wallet moves the money into the club's balance held by GameOn (paid out to the club later).
  IF @PayFromWallet=1 AND @Price>0
  BEGIN
   DECLARE @Neg decimal(12,2)=-@Price;
   EXEC dbo.WalletDebit @UserId,@Neg,'CourtPayment',@Id,NULL,N'Court booking';
   EXEC dbo.ClubLedgerPost @Club,@Price,'Sale','Booking',@Id;
  END;
  IF @Staff=1 EXEC dbo.LogStaff @Club,@UserId,'WalkInBooked',@GuestName;
  IF @Staff=0
  BEGIN
   DECLARE @Pts int=CASE WHEN dbo.IsPro(@UserId)=1 THEN 30 ELSE 20 END;
   EXEC dbo.LoyaltyPost @UserId,@Pts,'CourtBooking','Booking',@Id,N'Court booking';
   INSERT dbo.UserActivityLogs(UserId,SubtypeId,Action) SELECT @UserId,SubtypeId,'CourtBooked' FROM dbo.Courts WHERE CourtId=@CourtId;
   INSERT dbo.Notifications(UserId,Title,Body) SELECT s.UserId,IIF(@Status='Pending',N'Booking request to approve',N'New court booking'),LEFT(N'A player booked '+k.Name+N' at '+FORMAT(@Local,'ddd d MMM HH:mm')+IIF(@Status='Pending',N'. Approve or decline it in Club manager.',N''),800)
   FROM dbo.ClubStaff s JOIN dbo.Courts k ON k.ClubId=s.ClubId WHERE k.CourtId=@CourtId AND s.Role IN('Owner','Manager','Reception');
  END;
  COMMIT;
  SELECT BookingId,Price,ListPrice,Discount,StartsAt,EndsAt,Status,PaidFromWallet FROM dbo.CourtBookings WHERE BookingId=@Id;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.ReleaseCourtBooking @BookingId bigint,@NewStatus varchar(10),@ActorId int AS
BEGIN
 SET NOCOUNT ON;
 -- Caller owns the transaction and the booking lock. Returns wallet money, points and reward codes.
 DECLARE @Owner int,@Club int,@Paid bit,@Price decimal(10,2);
 SELECT @Owner=b.UserId,@Club=k.ClubId,@Paid=b.PaidFromWallet,@Price=b.Price FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE b.BookingId=@BookingId;
 UPDATE dbo.CourtBookings SET Status=@NewStatus,CancelledAt=IIF(@NewStatus='Cancelled',SYSUTCDATETIME(),CancelledAt),DecidedBy=IIF(@NewStatus='Declined',@ActorId,DecidedBy),DecidedAt=IIF(@NewStatus='Declined',SYSUTCDATETIME(),DecidedAt) WHERE BookingId=@BookingId;
 IF @Paid=1 AND @Price>0 AND EXISTS(SELECT 1 FROM dbo.WalletLedger WHERE Kind='CourtPayment' AND CourtBookingId=@BookingId)
 BEGIN
  EXEC dbo.WalletDebit @Owner,@Price,'CourtRefund',@BookingId,NULL,N'Court booking refunded to wallet';
  DECLARE @Neg decimal(12,2)=-@Price;
  EXEC dbo.ClubLedgerPost @Club,@Neg,'Refund','Booking',@BookingId;
 END;
 DECLARE @Earned int=(SELECT Points FROM dbo.LoyaltyLedger WHERE Reason='CourtBooking' AND RefType='Booking' AND RefId=@BookingId AND UserId=@Owner);
 IF @Owner IS NOT NULL AND @Earned>0 BEGIN SET @Earned=-@Earned; EXEC dbo.LoyaltyPost @Owner,@Earned,'Reversal','Booking',@BookingId,N'Court booking cancelled'; END;
 UPDATE x SET Status='Issued',UsedAt=NULL FROM dbo.RewardRedemptions x JOIN dbo.CourtBookings b ON b.RedemptionId=x.RedemptionId WHERE b.BookingId=@BookingId AND x.ExpiresAt>SYSUTCDATETIME();
END;
GO
CREATE OR ALTER PROCEDURE dbo.CancelCourtBooking @BookingId bigint,@UserId int AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Owner int,@Club int,@Start datetime2(0),@Status varchar(10);
  SELECT @Owner=b.UserId,@Club=k.ClubId,@Start=b.StartsAt,@Status=b.Status FROM dbo.CourtBookings b WITH(UPDLOCK,HOLDLOCK) JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE b.BookingId=@BookingId;
  IF @Status IS NULL THROW 50001,'Booking not found.',1;
  DECLARE @IsStaff bit=CASE WHEN dbo.ClubRoleRank(@Club,@UserId)>=2 THEN 1 ELSE 0 END;
  IF (@Owner IS NULL OR @Owner<>@UserId) AND @IsStaff=0 THROW 50003,'You can only cancel your own bookings.',1;
  IF @Status NOT IN('Confirmed','Pending') THROW 50004,'This booking is already closed.',1;
  -- Players can cancel up to 6 hours before (pending requests any time); staff can cancel any time.
  IF @IsStaff=0 AND @Status='Confirmed' AND @Start<DATEADD(hour,6,SYSUTCDATETIME()) THROW 50004,'Bookings can be cancelled up to 6 hours before. Call the club.',1;
  EXEC dbo.ReleaseCourtBooking @BookingId,'Cancelled',@UserId;
  IF @IsStaff=1 AND (@Owner IS NULL OR @Owner<>@UserId) EXEC dbo.LogStaff @Club,@UserId,'BookingCancelled',NULL;
  IF @Owner IS NOT NULL AND @Owner<>@UserId INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,N'Court booking cancelled by the club',N'The club cancelled your booking. Any wallet payment is back in your wallet. Contact them in Chat if you have questions.');
  COMMIT;
  SELECT 'Cancelled' AS Status;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.DecideCourtBooking @BookingId bigint,@ActorId int,@Approve bit AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Owner int,@Club int,@Status varchar(10),@Local datetime2(0),@Court nvarchar(80);
  SELECT @Owner=b.UserId,@Club=k.ClubId,@Status=b.Status,@Local=dbo.ToBeirut(b.StartsAt),@Court=k.Name FROM dbo.CourtBookings b WITH(UPDLOCK,HOLDLOCK) JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE b.BookingId=@BookingId;
  IF @Status IS NULL THROW 50001,'Booking not found.',1;
  IF dbo.ClubRoleRank(@Club,@ActorId)<2 THROW 50003,'Only club staff can approve bookings.',1;
  IF @Status<>'Pending' THROW 50004,'This request was already handled.',1;
  IF @Approve=1 UPDATE dbo.CourtBookings SET Status='Confirmed',DecidedBy=@ActorId,DecidedAt=SYSUTCDATETIME() WHERE BookingId=@BookingId;
  ELSE EXEC dbo.ReleaseCourtBooking @BookingId,'Declined',@ActorId;

  EXEC dbo.LogStaff @Club,@ActorId,@Act,@Court;
  INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,IIF(@Approve=1,N'Court booking confirmed',N'Court booking declined'),
   LEFT(@Court+N' at '+FORMAT(@Local,'ddd d MMM HH:mm')+IIF(@Approve=1,N' is confirmed. See you there!',N' could not be accepted. Any wallet payment was returned.'),800));
  COMMIT;
  SELECT IIF(@Approve=1,'Confirmed','Declined') AS Status;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;

CREATE OR ALTER PROCEDURE dbo.BuyClubProduct @ProductId int,@UserId int,@Key uniqueidentifier AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Id bigint=(SELECT PurchaseId FROM dbo.ClubPurchases WHERE UserId=@UserId AND IdempotencyKey=@Key);
  IF @Id IS NULL
  BEGIN
   DECLARE @Club int,@Kind varchar(10),@Price decimal(10,2),@Days smallint,@Credits smallint;
   SELECT @Club=ClubId,@Kind=Kind,@Price=Price,@Days=DurationDays,@Credits=Credits FROM dbo.ClubProducts WITH(UPDLOCK,HOLDLOCK) WHERE ProductId=@ProductId AND IsActive=1;
   IF @Club IS NULL THROW 50001,'This membership or pack is not on sale.',1;
   -- A renewed membership starts when the current one at the same club ends.
   DECLARE @Start datetime2(3)=SYSUTCDATETIME();
   IF @Kind='Membership' SET @Start=COALESCE((SELECT MAX(EndsAt) FROM dbo.ClubPurchases WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId AND ClubId=@Club AND Kind='Membership' AND Status='Active' AND EndsAt>SYSUTCDATETIME()),@Start);

   VALUES(@ProductId,@Club,@UserId,@Kind,@Price,@Start,DATEADD(day,@Days,@Start),@Credits,UPPER(LEFT(REPLACE(CONVERT(varchar(36),NEWID()),'-',''),12)),@Key);
   SET @Id=SCOPE_IDENTITY();
   DECLARE @Neg decimal(12,2)=-@Price;
   EXEC dbo.WalletDebit @UserId,@Neg,'ClubPurchase',NULL,@Id,N'Club membership / class pack';
   EXEC dbo.ClubLedgerPost @Club,@Price,'Sale','Purchase',@Id;
   INSERT dbo.Notifications(UserId,Title,Body) SELECT s.UserId,N'New member purchase',LEFT(N'A player bought '+p.Name+N'.',800) FROM dbo.ClubStaff s JOIN dbo.ClubProducts p ON p.ClubId=s.ClubId WHERE p.ProductId=@ProductId AND s.Role IN('Owner','Manager');
  END;
  COMMIT;
  SELECT x.PurchaseId,x.Kind,p.Name,x.StartsAt,x.EndsAt,x.CreditsLeft,x.AccessCode FROM dbo.ClubPurchases x JOIN dbo.ClubProducts p ON p.ProductId=x.ProductId WHERE x.PurchaseId=@Id;

CREATE OR ALTER PROCEDURE dbo.RegisterWithPack @ClassId int,@UserId int,@PurchaseId bigint AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Club int,@Cap smallint,@Start datetime2(0),@Status varchar(10);
  SELECT @Club=ClubId,@Cap=Capacity,@Start=StartsAt,@Status=Status FROM dbo.Classes WITH(UPDLOCK,HOLDLOCK) WHERE ClassId=@ClassId;
  IF @Club IS NULL THROW 50001,'Class not found.',1;
  IF @Status<>'Scheduled' OR @Start<=SYSUTCDATETIME() THROW 50002,'This class is no longer open.',1;
  DECLARE @Credits smallint;
  SELECT @Credits=CreditsLeft FROM dbo.ClubPurchases WITH(UPDLOCK,HOLDLOCK)
  WHERE PurchaseId=@PurchaseId AND UserId=@UserId AND ClubId=@Club AND Kind='ClassPack' AND Status='Active' AND StartsAt<=SYSUTCDATETIME() AND EndsAt>SYSUTCDATETIME();
  IF @Credits IS NULL THROW 50002,'Use a valid class pack from this club.',1;
  IF @Credits<1 THROW 50006,'Your class pack has no credits left.',1;
  IF EXISTS(SELECT 1 FROM dbo.ClassRegistrations WHERE ClassId=@ClassId AND UserId=@UserId AND Status='Registered') THROW 50004,'You are already registered.',1;
  -- A pack credit buys a confirmed place only: no credit is spent on a waitlist.
  IF (SELECT COUNT(*) FROM dbo.ClassRegistrations WHERE ClassId=@ClassId AND Status='Registered')>=@Cap THROW 50004,'This class is full. Join its waitlist without using a credit.',1;
  MERGE dbo.ClassRegistrations AS t USING (SELECT @ClassId c,@UserId u) s ON t.ClassId=s.c AND t.UserId=s.u
  WHEN MATCHED THEN UPDATE SET Status='Registered',PurchaseId=@PurchaseId,CreatedAt=SYSUTCDATETIME()
  WHEN NOT MATCHED THEN INSERT(ClassId,UserId,Status,PurchaseId) VALUES(@ClassId,@UserId,'Registered',@PurchaseId);
  UPDATE dbo.ClubPurchases SET CreditsLeft=CreditsLeft-1 WHERE PurchaseId=@PurchaseId;
  EXEC dbo.LoyaltyPost @UserId,15,'Class','Class',@ClassId,N'Class registration';
  INSERT dbo.UserActivityLogs(UserId,SubtypeId,Action) SELECT @UserId,SubtypeId,'ClassJoined' FROM dbo.Classes WHERE ClassId=@ClassId;
  COMMIT;
  SELECT 'Registered' AS Status,(SELECT CreditsLeft FROM dbo.ClubPurchases WHERE PurchaseId=@PurchaseId) AS CreditsLeft;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO

 UPDATE r SET PurchaseId=NULL FROM dbo.ClassRegistrations r JOIN inserted i ON i.ClassId=r.ClassId AND i.UserId=r.UserId JOIN deleted d ON d.ClassId=i.ClassId AND d.UserId=i.UserId

CREATE OR ALTER PROCEDURE dbo.ClubCustomers @ClubId int,@ActorId int AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.ClubRoleRank(@ClubId,@ActorId)<3 THROW 50003,'Managers and owners see customers.',1;
 -- CRM: everyone who booked, bought or trained here. Personal data shown is limited to name and activity.
 WITH people AS (
  SELECT b.UserId FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.UserId IS NOT NULL
  UNION SELECT r.UserId FROM dbo.ClassRegistrations r JOIN dbo.Classes c ON c.ClassId=r.ClassId WHERE c.ClubId=@ClubId
  UNION SELECT UserId FROM dbo.ClubPurchases WHERE ClubId=@ClubId)
 SELECT TOP(500) p.UserId,u.DisplayName,CAST(dbo.IsPro(p.UserId) AS bit) AS IsPro,
  (SELECT COUNT(*) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.UserId=p.UserId AND b.Status IN('Confirmed','Completed')) AS Bookings,
  (SELECT COUNT(*) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.UserId=p.UserId AND b.Status='Cancelled') AS Cancellations,
  (SELECT COUNT(*) FROM dbo.ClassRegistrations r JOIN dbo.Classes c ON c.ClassId=r.ClassId WHERE c.ClubId=@ClubId AND r.UserId=p.UserId AND r.Status='Registered') AS Classes,
  (SELECT COALESCE(SUM(b.Price),0) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.UserId=p.UserId AND b.Status IN('Confirmed','Completed'))
   +(SELECT COALESCE(SUM(Price),0) FROM dbo.ClubPurchases WHERE ClubId=@ClubId AND UserId=p.UserId AND Status='Active') AS Spend,
  (SELECT TOP(1) Name FROM dbo.ClubPurchases x JOIN dbo.ClubProducts pp ON pp.ProductId=x.ProductId WHERE x.ClubId=@ClubId AND x.UserId=p.UserId AND x.Kind='Membership' AND x.Status='Active' AND x.EndsAt>SYSUTCDATETIME() ORDER BY x.EndsAt DESC) AS ActiveMembership,
  (SELECT MAX(v) FROM (SELECT MAX(b.StartsAt) v FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.UserId=p.UserId AND b.Status IN('Confirmed','Completed') AND b.StartsAt<=SYSUTCDATETIME()
    UNION ALL SELECT MAX(CheckedInAt) FROM dbo.ClubCheckIns WHERE ClubId=@ClubId AND UserId=p.UserId) x) AS LastVisit
 FROM people p JOIN dbo.Users u ON u.UserId=p.UserId ORDER BY Spend DESC,u.DisplayName;
 SELECT TOP(10) Title,Recipients,SentAt FROM dbo.ClubBroadcasts WHERE ClubId=@ClubId ORDER BY BroadcastId DESC;
END;
GO

  SELECT @ClubId,@Salary+@Commission,'Staff',@PeriodMonth,LEFT(N'Payroll: '+u.DisplayName,200),@ActorId FROM dbo.Users u WHERE u.UserId=@UserId;

  EXEC dbo.LogStaff @ClubId,@ActorId,'Payroll',NULL;
  COMMIT;

  IF @ActorId<>@Owner THROW 50003,'Only the group owner adds members.',1;

  IF @UserId=@Owner THROW 50002,'The owner cannot leave; remove the others or keep the group.',1;

CREATE OR ALTER PROCEDURE dbo.InviteGroupToGame @GameId int,@HostUserId int,@GroupId int AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 BEGIN TRY
  BEGIN TRANSACTION;
  IF NOT EXISTS(SELECT 1 FROM dbo.GamesAndEvents WITH(UPDLOCK,HOLDLOCK) WHERE EventId=@GameId AND HostUserId=@HostUserId AND GameStatus='Open' AND StartsAt>SYSUTCDATETIME()) THROW 50003,'Only the host of an open game can invite a group.',1;
  IF NOT EXISTS(SELECT 1 FROM dbo.PlayerGroupMembers WHERE GroupId=@GroupId AND UserId=@HostUserId) THROW 50003,'You are not in this group.',1;
  -- Group members are pre-approved (works for approval and private games); each still confirms their own place.
  MERGE dbo.GameJoinRequests AS t USING (SELECT m.UserId FROM dbo.PlayerGroupMembers m WHERE m.GroupId=@GroupId AND m.UserId<>@HostUserId) s ON t.GameId=@GameId AND t.UserId=s.UserId
  WHEN MATCHED AND t.Status<>'Approved' THEN UPDATE SET Status='Approved',DecidedBy=@HostUserId,DecidedAt=SYSUTCDATETIME()
  WHEN NOT MATCHED THEN INSERT(GameId,UserId,Status,DecidedBy,DecidedAt) VALUES(@GameId,s.UserId,'Approved',@HostUserId,SYSUTCDATETIME());
  INSERT dbo.Notifications(UserId,GameId,Title,Body) SELECT m.UserId,@GameId,N'Your group is invited',N'The host invited your group to a game. Open it and confirm your place before the spots fill.'
  FROM dbo.PlayerGroupMembers m WHERE m.GroupId=@GroupId AND m.UserId<>@HostUserId AND NOT EXISTS(SELECT 1 FROM dbo.Participations p WHERE p.GameId=@GameId AND p.UserId=m.UserId);
  DECLARE @N int=@@ROWCOUNT;
  COMMIT;
  SELECT @N AS Invited;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER FUNCTION dbo.CanUseConversation(@ConversationId bigint,@UserId int) RETURNS bit AS
BEGIN
 DECLARE @Kind varchar(8),@Game int,@Club int,@L int,@H int,@Group int;
 SELECT @Kind=Kind,@Game=GameId,@Club=ClubId,@L=UserLow,@H=UserHigh,@Group=GroupId FROM dbo.Conversations WHERE ConversationId=@ConversationId;
 IF @Kind='Direct' RETURN CASE WHEN @UserId IN(@L,@H) AND EXISTS(SELECT 1 FROM dbo.Friendships WHERE UserLow=@L AND UserHigh=@H AND Status='Accepted') THEN 1 ELSE 0 END;
 IF @Kind='Game' RETURN CASE WHEN EXISTS(SELECT 1 FROM dbo.GamesAndEvents WHERE EventId=@Game AND HostUserId=@UserId)
  OR EXISTS(SELECT 1 FROM dbo.Participations WHERE GameId=@Game AND UserId=@UserId)
  OR EXISTS(SELECT 1 FROM dbo.GameJoinRequests WHERE GameId=@Game AND UserId=@UserId AND Status='Approved') THEN 1 ELSE 0 END;
 IF @Kind='Club' RETURN CASE WHEN @UserId=@L OR dbo.ClubRoleRank(@Club,@UserId)>=2 THEN 1 ELSE 0 END;
 IF @Kind='Group' RETURN CASE WHEN EXISTS(SELECT 1 FROM dbo.PlayerGroupMembers WHERE GroupId=@Group AND UserId=@UserId) THEN 1 ELSE 0 END;
 RETURN 0;
END;
GO
CREATE OR ALTER PROCEDURE dbo.ConversationMembers @ConversationId bigint AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @Kind varchar(8),@Game int,@Club int,@L int,@H int,@Group int;
 SELECT @Kind=Kind,@Game=GameId,@Club=ClubId,@L=UserLow,@H=UserHigh,@Group=GroupId FROM dbo.Conversations WHERE ConversationId=@ConversationId;
 IF @Kind='Direct' SELECT UserId FROM dbo.Users WHERE UserId IN(@L,@H);
 ELSE IF @Kind='Game' SELECT HostUserId AS UserId FROM dbo.GamesAndEvents WHERE EventId=@Game UNION SELECT UserId FROM dbo.Participations WHERE GameId=@Game UNION SELECT UserId FROM dbo.GameJoinRequests WHERE GameId=@Game AND Status='Approved';
 ELSE IF @Kind='Club' SELECT @L AS UserId UNION SELECT UserId FROM dbo.ClubStaff WHERE ClubId=@Club AND Role IN('Owner','Manager','Reception');
 ELSE IF @Kind='Group' SELECT UserId FROM dbo.PlayerGroupMembers WHERE GroupId=@Group;
END;
GO

CREATE OR ALTER PROCEDURE dbo.ShotAction @ShotId bigint,@UserId int,@Action varchar(8),@Reason nvarchar(200)=NULL AS
BEGIN
 SET NOCOUNT ON;
 DECLARE @Owner int=(SELECT UserId FROM dbo.Shots WHERE ShotId=@ShotId);
 IF @Owner IS NULL THROW 50001,'Shot not found.',1;
 IF @Action='Like' BEGIN IF NOT EXISTS(SELECT 1 FROM dbo.ShotLikes WHERE ShotId=@ShotId AND UserId=@UserId) INSERT dbo.ShotLikes(ShotId,UserId) VALUES(@ShotId,@UserId); END
 ELSE IF @Action='Unlike' DELETE dbo.ShotLikes WHERE ShotId=@ShotId AND UserId=@UserId;
 ELSE IF @Action='Report'
 BEGIN
  IF @Owner=@UserId THROW 50002,'You cannot report your own shot.',1;
  IF NULLIF(TRIM(@Reason),N'') IS NULL THROW 50002,'Say what is wrong.',1;
  IF NOT EXISTS(SELECT 1 FROM dbo.ShotReports WHERE ShotId=@ShotId AND UserId=@UserId) INSERT dbo.ShotReports(ShotId,UserId,Reason) VALUES(@ShotId,@UserId,TRIM(@Reason));
 END
 ELSE IF @Action='Delete'
 BEGIN
  -- Owners delete their own; GameOn operators can hide anything.
  IF @Owner<>@UserId AND NOT EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@UserId) THROW 50003,'You can only delete your own shots.',1;
  UPDATE dbo.Shots SET Hidden=1 WHERE ShotId=@ShotId;
 END
 ELSE THROW 50002,'Unknown action.',1;
 SELECT (SELECT COUNT(*) FROM dbo.ShotLikes WHERE ShotId=@ShotId) AS Likes;
END;
GO
CREATE OR ALTER PROCEDURE dbo.GetCourtSchedule @ClubId int,@Day date,@UserId int=NULL,@CourtId int=NULL,@IncludeClasses bit=1 AS
BEGIN
 SET NOCOUNT ON;
 -- @Day is a Beirut calendar day. Every slot is computed in Beirut time and returned with its UTC start.
 DECLARE @Rank int=dbo.ClubRoleRank(@ClubId,@UserId),@Pro bit=CASE WHEN @UserId IS NULL THEN 0 ELSE dbo.IsPro(@UserId) END;
 DECLARE @ProPct tinyint=(SELECT CASE WHEN IsPartner=1 THEN ProDiscountPct ELSE 0 END FROM dbo.Clubs WHERE ClubId=@ClubId);
 WITH n AS (SELECT TOP(48) ROW_NUMBER() OVER(ORDER BY (SELECT NULL))-1 AS i FROM sys.all_objects),
 slots AS (
  SELECT k.CourtId,k.Name AS Court,k.SubtypeId,k.SlotMinutes,DATEADD(minute,k.OpenHour*60+n.i*k.SlotMinutes,CAST(@Day AS datetime2(0))) AS LocalStart,k.BasePrice,k.PeakPrice,k.PeakStartHour,k.PeakEndHour
  FROM dbo.Courts k CROSS JOIN n WHERE k.ClubId=@ClubId AND k.IsActive=1 AND (@CourtId IS NULL OR k.CourtId=@CourtId) AND k.OpenHour*60+(n.i+1)*k.SlotMinutes<=k.CloseHour*60)
 SELECT s.CourtId,s.Court,s.SubtypeId,s.LocalStart,DATEADD(minute,s.SlotMinutes,s.LocalStart) AS LocalEnd,dbo.FromBeirut(s.LocalStart) AS StartsAtUtc,
  CAST(CASE WHEN DATEPART(hour,s.LocalStart)>=s.PeakStartHour AND DATEPART(hour,s.LocalStart)<s.PeakEndHour THEN 1 ELSE 0 END AS bit) AS IsPeak,
  CASE WHEN DATEPART(hour,s.LocalStart)>=s.PeakStartHour AND DATEPART(hour,s.LocalStart)<s.PeakEndHour THEN s.PeakPrice ELSE s.BasePrice END AS ListPrice,
  CASE WHEN @Pro=1 THEN @ProPct ELSE 0 END AS ProDiscountPct,
  CASE WHEN dbo.FromBeirut(s.LocalStart)<=SYSUTCDATETIME() THEN 'Past'
   WHEN EXISTS(SELECT 1 FROM dbo.CourtBlocks b WHERE b.CourtId=s.CourtId AND b.StartsAt<dbo.FromBeirut(DATEADD(minute,s.SlotMinutes,s.LocalStart)) AND b.EndsAt>dbo.FromBeirut(s.LocalStart)) THEN 'Blocked'
   WHEN EXISTS(SELECT 1 FROM dbo.CourtBookings cb WHERE cb.CourtId=s.CourtId AND cb.Status IN('Pending','Confirmed','Completed','NoShow') AND cb.StartsAt<dbo.FromBeirut(DATEADD(minute,s.SlotMinutes,s.LocalStart)) AND cb.EndsAt>dbo.FromBeirut(s.LocalStart))
    THEN CASE WHEN EXISTS(SELECT 1 FROM dbo.CourtBookings cb WHERE cb.CourtId=s.CourtId AND cb.UserId=@UserId AND cb.Status IN('Pending','Confirmed') AND cb.StartsAt=dbo.FromBeirut(s.LocalStart)) THEN 'Mine' WHEN @Rank>=2 AND EXISTS(SELECT 1 FROM dbo.CourtBookings cb WHERE cb.CourtId=s.CourtId AND cb.Status='Pending' AND cb.StartsAt=dbo.FromBeirut(s.LocalStart)) THEN 'Pending' ELSE 'Booked' END
   ELSE 'Free' END AS SlotStatus,
  -- Staff see who booked and why a slot is blocked; players never see other players' names.
  CASE WHEN @Rank>=2 THEN (SELECT TOP(1) COALESCE(u.DisplayName,cb.GuestName) FROM dbo.CourtBookings cb LEFT JOIN dbo.Users u ON u.UserId=cb.UserId WHERE cb.CourtId=s.CourtId AND cb.Status IN('Pending','Confirmed','Completed','NoShow') AND cb.StartsAt=dbo.FromBeirut(s.LocalStart)) END AS BookedBy,
  CASE WHEN @Rank>=2 THEN (SELECT TOP(1) cb.BookingId FROM dbo.CourtBookings cb WHERE cb.CourtId=s.CourtId AND cb.Status IN('Pending','Confirmed','Completed','NoShow') AND cb.StartsAt=dbo.FromBeirut(s.LocalStart))
   ELSE (SELECT TOP(1) cb.BookingId FROM dbo.CourtBookings cb WHERE cb.CourtId=s.CourtId AND cb.UserId=@UserId AND cb.Status IN('Pending','Confirmed') AND cb.StartsAt=dbo.FromBeirut(s.LocalStart)) END AS BookingId,
  CASE WHEN @Rank>=2 THEN (SELECT TOP(1) b.BlockId FROM dbo.CourtBlocks b WHERE b.CourtId=s.CourtId AND b.StartsAt<=dbo.FromBeirut(s.LocalStart) AND b.EndsAt>=dbo.FromBeirut(DATEADD(minute,s.SlotMinutes,s.LocalStart))) END AS BlockId,
  CASE WHEN @Rank>=2 THEN (SELECT TOP(1) b.Reason FROM dbo.CourtBlocks b WHERE b.CourtId=s.CourtId AND b.StartsAt<dbo.FromBeirut(DATEADD(minute,s.SlotMinutes,s.LocalStart)) AND b.EndsAt>dbo.FromBeirut(s.LocalStart)) END AS BlockReason
 FROM slots s ORDER BY s.CourtId,s.LocalStart;
 IF @IncludeClasses=1
 SELECT ClassId,Title,s.Name AS Sport,dbo.ToBeirut(StartsAt) AS LocalStart,DurationMinutes,Capacity,Price,CourtId,
  (SELECT COUNT(*) FROM dbo.ClassRegistrations r WHERE r.ClassId=c.ClassId AND r.Status='Registered') AS Registered,
  (SELECT Status FROM dbo.ClassRegistrations r WHERE r.ClassId=c.ClassId AND r.UserId=@UserId AND r.Status<>'Cancelled') AS MyStatus,
  (SELECT DisplayName FROM dbo.Users u WHERE u.UserId=c.CoachUserId) AS Coach
 FROM dbo.Classes c JOIN dbo.Subtypes s ON s.SubtypeId=c.SubtypeId
 WHERE c.ClubId=@ClubId AND c.Status='Scheduled' AND dbo.ToBeirut(c.StartsAt)>=CAST(@Day AS datetime2(0)) AND dbo.ToBeirut(c.StartsAt)<DATEADD(day,1,CAST(@Day AS datetime2(0))) ORDER BY c.StartsAt;
END;
GO

CREATE OR ALTER PROCEDURE dbo.GetClubDashboard @ClubId int,@ActorId int,@Days int=30 AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.ClubRoleRank(@ClubId,@ActorId)<3 THROW 50003,'Managers and owners can see the dashboard.',1;
 IF @Days NOT BETWEEN 1 AND 365 SET @Days=30;
 DECLARE @From datetime2(0)=DATEADD(day,-@Days,SYSUTCDATETIME()),@Now datetime2(0)=SYSUTCDATETIME();
 -- Revenue = court bookings that have taken place (not cancelled) + class registrations, at the price the player owes the club.
 DECLARE @Court decimal(14,2)=(SELECT COALESCE(SUM(b.Price),0) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.Status IN('Confirmed','Completed') AND b.StartsAt>=@From AND b.StartsAt<@Now);
 DECLARE @Class decimal(14,2)=(SELECT COALESCE(SUM(c.Price),0) FROM dbo.ClassRegistrations r JOIN dbo.Classes c ON c.ClassId=r.ClassId WHERE c.ClubId=@ClubId AND r.Status='Registered' AND c.Status='Scheduled' AND c.StartsAt>=@From AND c.StartsAt<@Now);
 DECLARE @Exp decimal(14,2)=(SELECT COALESCE(SUM(Amount),0) FROM dbo.ClubExpenses WHERE ClubId=@ClubId AND SpentOn>=CAST(@From AS date));
 DECLARE @SlotsTotal decimal(14,2)=(SELECT COALESCE(SUM(CAST((CloseHour-OpenHour)*60/SlotMinutes AS decimal(14,2))),0)*@Days FROM dbo.Courts WHERE ClubId=@ClubId AND IsActive=1);
 DECLARE @SlotsUsed decimal(14,2)=(SELECT COUNT(*) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.Status IN('Confirmed','Completed') AND b.StartsAt>=@From AND b.StartsAt<@Now);
 SELECT @Court AS CourtRevenue,@Class AS ClassRevenue,@Court+@Class AS Revenue,@Exp AS Expenses,@Court+@Class-@Exp AS NetProfit,
  CAST(CASE WHEN @SlotsTotal=0 THEN 0 ELSE 100*@SlotsUsed/@SlotsTotal END AS decimal(5,1)) AS OccupancyPct,
  (SELECT COUNT(*) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.Status='Confirmed' AND b.StartsAt>=@Now) AS UpcomingBookings,
  (SELECT COUNT(*) FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.Status='Cancelled' AND b.StartsAt>=@From) AS Cancellations,
  @Days AS Days;
 SELECT CAST(dbo.ToBeirut(b.StartsAt) AS date) AS Day,SUM(b.Price) AS Revenue,COUNT(*) AS Bookings
 FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId WHERE k.ClubId=@ClubId AND b.Status IN('Confirmed','Completed') AND b.StartsAt>=@From AND b.StartsAt<@Now
 GROUP BY CAST(dbo.ToBeirut(b.StartsAt) AS date) ORDER BY Day;
 SELECT DATEPART(hour,dbo.ToBeirut(b.StartsAt)) AS Hour,COUNT(*) AS Bookings FROM dbo.CourtBookings b JOIN dbo.Courts k ON k.CourtId=b.CourtId
 WHERE k.ClubId=@ClubId AND b.Status IN('Confirmed','Completed') AND b.StartsAt>=@From GROUP BY DATEPART(hour,dbo.ToBeirut(b.StartsAt)) ORDER BY Hour;
 SELECT Category,SUM(Amount) AS Amount FROM dbo.ClubExpenses WHERE ClubId=@ClubId AND SpentOn>=CAST(@From AS date) GROUP BY Category ORDER BY Amount DESC;
 SELECT k.CourtId,k.Name,k.SubtypeId,k.SlotMinutes,k.OpenHour,k.CloseHour,k.BasePrice,k.PeakPrice,k.PeakStartHour,k.PeakEndHour,k.IsActive,k.RequiresApproval FROM dbo.Courts k WHERE k.ClubId=@ClubId ORDER BY k.Name;
 SELECT s.UserId,u.DisplayName,u.Email,s.Role FROM dbo.ClubStaff s JOIN dbo.Users u ON u.UserId=s.UserId WHERE s.ClubId=@ClubId AND dbo.ClubRoleRank(@ClubId,@ActorId)>=4 ORDER BY s.Role,u.DisplayName;
END;
GO
