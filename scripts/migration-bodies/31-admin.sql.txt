-- Non-executable template. scripts/build-migration.py turns this into database/31-admin.sql.
-- Platform admins: user management (suspend/restore/delete with anonymisation), roles, app settings,
-- moderation and an admin audit log. Adds a per-user session stamp so old cookies can never map to another account.
USE GameOn;
GO
BEGIN TRANSACTION;
CREATE TABLE dbo.Admins(UserId int NOT NULL CONSTRAINT PK_Admins PRIMARY KEY REFERENCES dbo.Users(UserId),
 AddedBy int NULL REFERENCES dbo.Users(UserId), AddedAt datetime2(3) NOT NULL CONSTRAINT DF_Admins_Added DEFAULT SYSUTCDATETIME());
ALTER TABLE dbo.Users ADD AccountStatus varchar(10) NOT NULL CONSTRAINT DF_Users_Status DEFAULT 'Active'
  CONSTRAINT CK_Users_Status CHECK(AccountStatus IN('Active','Suspended','Deleted')),
 StatusReason nvarchar(300) NULL, StatusChangedAt datetime2(3) NULL, StatusChangedBy int NULL REFERENCES dbo.Users(UserId),
 SessionStamp uniqueidentifier NOT NULL CONSTRAINT DF_Users_Stamp DEFAULT NEWID(), LastLoginAt datetime2(3) NULL;
CREATE INDEX IX_Users_Status ON dbo.Users(AccountStatus) INCLUDE(DisplayName,Email,CreatedAt);
CREATE TABLE dbo.AdminAudit(AuditId bigint IDENTITY CONSTRAINT PK_AdminAudit PRIMARY KEY, AdminId int NOT NULL REFERENCES dbo.Users(UserId),
 Action varchar(24) NOT NULL, TargetUserId int NULL REFERENCES dbo.Users(UserId), Detail nvarchar(400) NULL,
 At datetime2(3) NOT NULL CONSTRAINT DF_AdminAudit_At DEFAULT SYSUTCDATETIME());
CREATE INDEX IX_AdminAudit_At ON dbo.AdminAudit(AuditId DESC);
CREATE TABLE dbo.PlatformSettings(SettingKey varchar(40) NOT NULL CONSTRAINT PK_PlatformSettings PRIMARY KEY, SettingValue nvarchar(400) NOT NULL,
 UpdatedBy int NULL REFERENCES dbo.Users(UserId), UpdatedAt datetime2(3) NOT NULL CONSTRAINT DF_Settings_Updated DEFAULT SYSUTCDATETIME(),
 CONSTRAINT CK_Settings_Key CHECK(SettingKey IN('Announcement','SignupsOpen','ShotsEnabled','ProMonthlyPrice','SupportEmail')));
INSERT dbo.PlatformSettings(SettingKey,SettingValue) VALUES('Announcement',N''),('SignupsOpen',N'true'),('ShotsEnabled',N'true'),('ProMonthlyPrice',N'5.00'),('SupportEmail',N'');
GO
CREATE OR ALTER FUNCTION dbo.IsAdmin(@UserId int) RETURNS bit AS
BEGIN
 RETURN CASE WHEN EXISTS(SELECT 1 FROM dbo.Admins a JOIN dbo.Users u ON u.UserId=a.UserId WHERE a.UserId=@UserId AND u.AccountStatus='Active') THEN 1 ELSE 0 END;
END;
GO
CREATE OR ALTER PROCEDURE dbo.FindUserForLogin @Email nvarchar(254) AS
BEGIN
 SET NOCOUNT ON;
 SELECT UserId,DisplayName,PasswordHash,SecurityVersion,AccountStatus,SessionStamp FROM dbo.Users WHERE Email=@Email;
END;
GO
CREATE OR ALTER PROCEDURE dbo.CheckSession @UserId int AS
BEGIN
 SET NOCOUNT ON;
 -- Suspended or deleted accounts have no valid session; the stamp ties a cookie to one specific account row.
 SELECT UserId,SecurityVersion,SessionStamp FROM dbo.Users WHERE UserId=@UserId AND AccountStatus='Active';
END;
GO
CREATE OR ALTER PROCEDURE dbo.RecordLogin @UserId int AS
BEGIN
 SET NOCOUNT ON;
 UPDATE dbo.Users SET LastLoginAt=SYSUTCDATETIME() WHERE UserId=@UserId;
END;
GO
CREATE OR ALTER PROCEDURE dbo.GetProfile @UserId int AS
BEGIN
 SET NOCOUNT ON;
 SELECT u.UserId,u.Email,u.DisplayName,u.Bio,u.Gender,u.DateOfBirth,p.Area,COALESCE(p.SkillLevel,'All levels') AS SkillLevel,
  CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.PaymentOperators o WHERE o.UserId=u.UserId) THEN 1 ELSE 0 END AS bit) AS CanManageVenues,
  dbo.IsAdmin(u.UserId) AS IsAdmin
 FROM dbo.Users u LEFT JOIN dbo.UserPreferences p ON p.UserId=u.UserId WHERE u.UserId=@UserId;
 SELECT SportId FROM dbo.UserSportPreferences WHERE UserId=@UserId ORDER BY SportId;
END;
GO
CREATE OR ALTER PROCEDURE dbo.GetPlatformSettings AS
BEGIN
 SET NOCOUNT ON;
 SELECT SettingKey,SettingValue,UpdatedAt FROM dbo.PlatformSettings;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminSetSetting @AdminId int,@Key varchar(40),@Value nvarchar(400) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 SET @Value=TRIM(COALESCE(@Value,N''));
 IF @Key IN('SignupsOpen','ShotsEnabled') AND @Value NOT IN(N'true',N'false') THROW 50002,'Use true or false.',1;
 IF @Key='ProMonthlyPrice' AND (TRY_CONVERT(decimal(10,2),@Value) IS NULL OR TRY_CONVERT(decimal(10,2),@Value) NOT BETWEEN 1 AND 100) THROW 50002,'Pro price must be between 1 and 100 USD.',1;
 IF @Key='SupportEmail' AND @Value<>N'' AND @Value NOT LIKE N'_%@_%._%' THROW 50002,'Enter a valid support email or leave it empty.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  UPDATE dbo.PlatformSettings SET SettingValue=@Value,UpdatedBy=@AdminId,UpdatedAt=SYSUTCDATETIME() WHERE SettingKey=@Key;
  IF @@ROWCOUNT=0 THROW 50002,'Unknown setting.',1;
  INSERT dbo.AdminAudit(AdminId,Action,Detail) VALUES(@AdminId,'Setting',LEFT(@Key+N' = '+@Value,400));
  COMMIT;
  SELECT SettingKey,SettingValue FROM dbo.PlatformSettings WHERE SettingKey=@Key;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminStats @AdminId int AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 SELECT (SELECT COUNT(*) FROM dbo.Users WHERE AccountStatus='Active' AND PasswordHash<>'DISABLED_FIXTURE') AS ActiveUsers,
  (SELECT COUNT(*) FROM dbo.Users WHERE AccountStatus='Suspended') AS SuspendedUsers,
  (SELECT COUNT(*) FROM dbo.Users WHERE CreatedAt>DATEADD(day,-7,SYSUTCDATETIME()) AND PasswordHash<>'DISABLED_FIXTURE') AS NewUsers7d,
  (SELECT COUNT(*) FROM dbo.Users WHERE LastLoginAt>DATEADD(day,-7,SYSUTCDATETIME())) AS ActiveLast7d,
  (SELECT COUNT(*) FROM dbo.GamesAndEvents WHERE GameStatus='Open' AND StartsAt>SYSUTCDATETIME()) AS UpcomingGames,
  (SELECT COUNT(*) FROM dbo.CourtBookings WHERE Status IN('Confirmed','Pending') AND StartsAt>SYSUTCDATETIME()) AS UpcomingCourtBookings,
  (SELECT COUNT(*) FROM dbo.Clubs WHERE IsActive=1) AS Clubs,
  (SELECT COALESCE(SUM(Balance),0) FROM dbo.WalletAccounts) AS WalletMoneyHeld,
  (SELECT COALESCE(SUM(Balance),0) FROM dbo.Clubs) AS ClubMoneyHeld,
  (SELECT COUNT(*) FROM dbo.WalletTopUps WHERE Status='Submitted') AS TopUpsToCheck,
  (SELECT COUNT(*) FROM dbo.DepositDisputes WHERE Status='Open') AS OpenDisputes,
  (SELECT COUNT(DISTINCT ShotId) FROM dbo.ShotReports r WHERE EXISTS(SELECT 1 FROM dbo.Shots s WHERE s.ShotId=r.ShotId AND s.Hidden=0)) AS ReportedShots;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminListUsers @AdminId int,@Query nvarchar(100)=NULL,@Status varchar(10)=NULL AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 DECLARE @Q nvarchar(100)=NULLIF(TRIM(@Query),N'');
 SELECT TOP(200) u.UserId,u.DisplayName,u.Email,u.CreatedAt,u.LastLoginAt,u.AccountStatus,u.StatusReason,
  dbo.IsAdmin(u.UserId) AS IsAdmin,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.PaymentOperators o WHERE o.UserId=u.UserId) THEN 1 ELSE 0 END AS bit) AS IsOperator,
  CAST(dbo.IsPro(u.UserId) AS bit) AS IsPro,COALESCE(w.Balance,0) AS WalletBalance,
  (SELECT COUNT(*) FROM dbo.GamesAndEvents g WHERE g.HostUserId=u.UserId) AS GamesHosted,
  (SELECT COUNT(*) FROM dbo.Participations p WHERE p.UserId=u.UserId) AS GamesJoined,
  (SELECT COUNT(*) FROM dbo.ShotReports r JOIN dbo.Shots s ON s.ShotId=r.ShotId WHERE s.UserId=u.UserId) AS ReportsAgainst,
  (SELECT STRING_AGG(c.Name+N' ('+s.Role+N')',N', ') FROM dbo.ClubStaff s JOIN dbo.Clubs c ON c.ClubId=s.ClubId WHERE s.UserId=u.UserId) AS Clubs
 FROM dbo.Users u LEFT JOIN dbo.WalletAccounts w ON w.UserId=u.UserId
 WHERE u.PasswordHash<>'DISABLED_FIXTURE' AND (@Status IS NULL OR u.AccountStatus=@Status)
  AND (@Q IS NULL OR CHARINDEX(@Q,u.DisplayName)>0 OR CHARINDEX(@Q,u.Email)>0 OR CONVERT(nvarchar(12),u.UserId)=@Q)
 ORDER BY u.UserId DESC;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminSetUserStatus @AdminId int,@UserId int,@Status varchar(10),@Reason nvarchar(300) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 IF @Status NOT IN('Active','Suspended') THROW 50002,'Choose Active or Suspended.',1;
 IF @UserId=@AdminId THROW 50002,'You cannot change your own account status.',1;
 IF @Status='Suspended' AND (@Reason IS NULL OR LEN(TRIM(@Reason))<4) THROW 50002,'Write the reason for the suspension.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Old varchar(10)=(SELECT AccountStatus FROM dbo.Users WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId);
  IF @Old IS NULL THROW 50001,'User not found.',1;
  IF @Old='Deleted' THROW 50004,'Deleted accounts cannot be restored.',1;
  IF @Status='Suspended' AND dbo.IsAdmin(@UserId)=1 THROW 50004,'Remove the admin role before suspending this account.',1;
  -- Suspending changes the session stamp and security version, so every signed-in device is logged out immediately.
  UPDATE dbo.Users SET AccountStatus=@Status,StatusReason=IIF(@Status='Suspended',TRIM(@Reason),NULL),StatusChangedAt=SYSUTCDATETIME(),StatusChangedBy=@AdminId,
   SecurityVersion=SecurityVersion+IIF(@Status='Suspended',1,0),SessionStamp=IIF(@Status='Suspended',NEWID(),SessionStamp) WHERE UserId=@UserId;
  INSERT dbo.AdminAudit(AdminId,Action,TargetUserId,Detail) VALUES(@AdminId,IIF(@Status='Suspended','Suspend','Restore'),@UserId,TRIM(@Reason));
  COMMIT;
  SELECT @Status AS AccountStatus;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminDeleteUser @AdminId int,@UserId int,@Reason nvarchar(300) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 IF @UserId=@AdminId THROW 50002,'You cannot delete your own account here.',1;
 IF @Reason IS NULL OR LEN(TRIM(@Reason))<4 THROW 50002,'Write the reason for deleting this account.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Status varchar(10)=(SELECT AccountStatus FROM dbo.Users WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId);
  IF @Status IS NULL THROW 50001,'User not found.',1;
  IF @Status='Deleted' THROW 50004,'This account is already deleted.',1;
  IF dbo.IsAdmin(@UserId)=1 THROW 50004,'Remove the admin role before deleting this account.',1;
  -- Never make money disappear: the player must withdraw (or an operator refund) first.
  IF EXISTS(SELECT 1 FROM dbo.WalletAccounts WHERE UserId=@UserId AND Balance>0) THROW 50004,'This user still has money in their wallet. Send it back (withdrawal) before deleting.',1;
  IF EXISTS(SELECT 1 FROM dbo.WalletWithdrawals WHERE UserId=@UserId AND Status='Requested') THROW 50004,'This user has a withdrawal waiting. Process it first.',1;
  IF EXISTS(SELECT 1 FROM dbo.DepositBookings WHERE UserId=@UserId AND DepositStatus IN('PENDING','PAID_HELD','REFUND_PENDING','DISPUTED')) THROW 50004,'This user has open deposits. Settle them first.',1;
  IF EXISTS(SELECT 1 FROM dbo.ClubStaff WHERE UserId=@UserId AND Role='Owner') THROW 50004,'This user owns a club. Give the club a new owner first.',1;
  -- Cancel their upcoming hosted games (players are notified and refunded by the normal rules).
  DECLARE @G int;
  DECLARE g CURSOR LOCAL FAST_FORWARD FOR SELECT EventId FROM dbo.GamesAndEvents WHERE HostUserId=@UserId AND GameStatus='Open' AND StartsAt>SYSUTCDATETIME();
  OPEN g; FETCH NEXT FROM g INTO @G;
  WHILE @@FETCH_STATUS=0 BEGIN EXEC dbo.CancelGame @GameId=@G,@HostUserId=@UserId; FETCH NEXT FROM g INTO @G; END;
  CLOSE g; DEALLOCATE g;
  -- Leave upcoming games and waitlists, cancel upcoming court bookings and classes.
  DELETE w FROM dbo.Waitlists w JOIN dbo.GamesAndEvents e ON e.EventId=w.GameId WHERE w.UserId=@UserId AND e.StartsAt>SYSUTCDATETIME();
  DELETE p FROM dbo.Participations p JOIN dbo.GamesAndEvents e ON e.EventId=p.GameId WHERE p.UserId=@UserId AND e.StartsAt>SYSUTCDATETIME() AND e.HostUserId<>@UserId;
  UPDATE dbo.CourtBookings SET Status='Cancelled',CancelledAt=SYSUTCDATETIME() WHERE UserId=@UserId AND Status IN('Pending','Confirmed') AND StartsAt>SYSUTCDATETIME() AND PaidFromWallet=0;
  IF EXISTS(SELECT 1 FROM dbo.CourtBookings WHERE UserId=@UserId AND Status IN('Pending','Confirmed') AND StartsAt>SYSUTCDATETIME()) THROW 50004,'This user has paid court bookings coming up. Cancel them (refund) first.',1;
  UPDATE r SET Status='Cancelled' FROM dbo.ClassRegistrations r JOIN dbo.Classes c ON c.ClassId=r.ClassId WHERE r.UserId=@UserId AND r.Status<>'Cancelled' AND c.StartsAt>SYSUTCDATETIME();
  -- Erase personal data; keep the anonymous row so bookings, payments and ledgers stay consistent.
  DELETE dbo.UserSportPreferences WHERE UserId=@UserId;
  UPDATE dbo.UserPreferences SET Area=NULL WHERE UserId=@UserId;
  DELETE dbo.Friendships WHERE @UserId IN(UserLow,UserHigh);
  DELETE dbo.PlayerGroupMembers WHERE UserId=@UserId;
  DELETE dbo.ClubStaff WHERE UserId=@UserId;
  DELETE dbo.PaymentOperators WHERE UserId=@UserId;
  DELETE dbo.CoachSports WHERE UserId=@UserId;
  DELETE dbo.CoachProfiles WHERE UserId=@UserId;
  UPDATE dbo.Shots SET Hidden=1,Caption=N'' WHERE UserId=@UserId;
  UPDATE dbo.Messages SET Body=N'(message from a deleted account)' WHERE SenderId=@UserId;
  UPDATE dbo.Users SET AccountStatus='Deleted',DisplayName=N'Deleted user',Email=N'deleted-'+CONVERT(nvarchar(12),@UserId)+N'@gameon.invalid',
   PasswordHash='DISABLED_FIXTURE',Bio=N'',Gender=NULL,DateOfBirth=NULL,StatusReason=TRIM(@Reason),StatusChangedAt=SYSUTCDATETIME(),StatusChangedBy=@AdminId,
   SecurityVersion=SecurityVersion+1,SessionStamp=NEWID() WHERE UserId=@UserId;
  INSERT dbo.AdminAudit(AdminId,Action,TargetUserId,Detail) VALUES(@AdminId,'Delete',@UserId,TRIM(@Reason));
  COMMIT;
  SELECT 'Deleted' AS AccountStatus;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminSetRole @AdminId int,@UserId int,@Role varchar(10),@Grant bit AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 IF @Role NOT IN('Admin','Operator') THROW 50002,'Choose Admin or Operator.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  IF NOT EXISTS(SELECT 1 FROM dbo.Users WHERE UserId=@UserId AND AccountStatus='Active') THROW 50004,'Only active accounts can get roles.',1;
  IF @Role='Admin'
  BEGIN
   IF @Grant=1 AND NOT EXISTS(SELECT 1 FROM dbo.Admins WHERE UserId=@UserId) INSERT dbo.Admins(UserId,AddedBy) VALUES(@UserId,@AdminId);
   IF @Grant=0
   BEGIN
    IF (SELECT COUNT(*) FROM dbo.Admins a JOIN dbo.Users u ON u.UserId=a.UserId WHERE u.AccountStatus='Active' AND a.UserId<>@UserId)=0 THROW 50004,'GameOn must keep at least one admin.',1;
    DELETE dbo.Admins WHERE UserId=@UserId;
   END;
  END
  ELSE
  BEGIN
   IF @Grant=1 AND NOT EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@UserId) INSERT dbo.PaymentOperators(UserId) VALUES(@UserId);
   IF @Grant=0 DELETE dbo.PaymentOperators WHERE UserId=@UserId;
  END;
  INSERT dbo.AdminAudit(AdminId,Action,TargetUserId,Detail) VALUES(@AdminId,IIF(@Grant=1,'Grant','Revoke'),@UserId,@Role);
  COMMIT;
  SELECT @Role AS Role,@Grant AS Granted;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminReportedShots @AdminId int AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 SELECT TOP(100) s.ShotId,s.Caption,s.Hidden,u.UserId,u.DisplayName,s.CreatedAt,COUNT(r.UserId) AS Reports,STRING_AGG(LEFT(r.Reason,80),N' | ') AS Reasons
 FROM dbo.Shots s JOIN dbo.ShotReports r ON r.ShotId=s.ShotId JOIN dbo.Users u ON u.UserId=s.UserId
 GROUP BY s.ShotId,s.Caption,s.Hidden,u.UserId,u.DisplayName,s.CreatedAt ORDER BY s.Hidden,COUNT(r.UserId) DESC,s.ShotId DESC;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminModerateShot @AdminId int,@ShotId bigint,@Hide bit AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 UPDATE dbo.Shots SET Hidden=@Hide WHERE ShotId=@ShotId;
 IF @@ROWCOUNT=0 THROW 50001,'Shot not found.',1;
 -- Restoring a wrongly reported photo clears its reports.
 IF @Hide=0 DELETE dbo.ShotReports WHERE ShotId=@ShotId;
 INSERT dbo.AdminAudit(AdminId,Action,Detail) VALUES(@AdminId,IIF(@Hide=1,'HideShot','RestoreShot'),CONVERT(nvarchar(20),@ShotId));
 SELECT @Hide AS Hidden;
END;
GO
CREATE OR ALTER PROCEDURE dbo.AdminAuditLog @AdminId int AS
BEGIN
 SET NOCOUNT ON;
 IF dbo.IsAdmin(@AdminId)=0 THROW 50003,'Admins only.',1;
 SELECT TOP(200) a.AuditId,ad.DisplayName AS AdminName,a.Action,t.DisplayName AS TargetName,a.TargetUserId,a.Detail,a.At
 FROM dbo.AdminAudit a JOIN dbo.Users ad ON ad.UserId=a.AdminId LEFT JOIN dbo.Users t ON t.UserId=a.TargetUserId ORDER BY a.AuditId DESC;
END;
GO
