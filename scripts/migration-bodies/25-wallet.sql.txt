-- ===== Wallet: append-only ledger + cached balance =====
CREATE TABLE dbo.WalletAccounts(UserId int NOT NULL CONSTRAINT PK_WalletAccounts PRIMARY KEY REFERENCES dbo.Users(UserId),
 Balance decimal(12,2) NOT NULL CONSTRAINT DF_Wallet_Balance DEFAULT 0 CONSTRAINT CK_Wallet_Balance CHECK(Balance>=0 AND Balance<=500),
 UpdatedAt datetime2(3) NOT NULL CONSTRAINT DF_Wallet_Updated DEFAULT SYSUTCDATETIME());
CREATE TABLE dbo.WalletTopUps(TopUpId uniqueidentifier NOT NULL CONSTRAINT PK_WalletTopUps PRIMARY KEY, UserId int NOT NULL REFERENCES dbo.Users(UserId),
 IdempotencyKey uniqueidentifier NOT NULL, Amount decimal(12,2) NOT NULL CONSTRAINT CK_TopUp_Amount CHECK(Amount BETWEEN 1 AND 200 AND Amount=ROUND(Amount,2)),
 Currency char(3) NOT NULL CONSTRAINT DF_TopUp_Currency DEFAULT 'USD' CONSTRAINT CK_TopUp_Currency CHECK(Currency='USD'), RecipientNumber varchar(20) NOT NULL,
 TransferReference nvarchar(100) NULL, Status varchar(20) NOT NULL CONSTRAINT DF_TopUp_Status DEFAULT 'AwaitingReference'
 CONSTRAINT CK_TopUp_Status CHECK(Status IN('AwaitingReference','Submitted','Verified','Rejected','Cancelled')),
 ReviewedBy int NULL REFERENCES dbo.Users(UserId), ReviewNote nvarchar(400) NULL, CreatedAt datetime2(3) NOT NULL CONSTRAINT DF_TopUp_Created DEFAULT SYSUTCDATETIME(), ReviewedAt datetime2(3) NULL,
 CONSTRAINT UQ_TopUp_Key UNIQUE(UserId,IdempotencyKey));
CREATE UNIQUE INDEX UX_TopUp_Reference ON dbo.WalletTopUps(TransferReference) WHERE TransferReference IS NOT NULL;
CREATE INDEX IX_TopUp_Status ON dbo.WalletTopUps(Status,CreatedAt) INCLUDE(UserId,Amount,TransferReference);
CREATE TABLE dbo.WalletWithdrawals(WithdrawalId uniqueidentifier NOT NULL CONSTRAINT PK_WalletWithdrawals PRIMARY KEY, UserId int NOT NULL REFERENCES dbo.Users(UserId),
 IdempotencyKey uniqueidentifier NOT NULL, Amount decimal(12,2) NOT NULL CONSTRAINT CK_Withdrawal_Amount CHECK(Amount BETWEEN 1 AND 500 AND Amount=ROUND(Amount,2)),
 WhishNumber varchar(20) NOT NULL CONSTRAINT CK_Withdrawal_Number CHECK(WhishNumber LIKE '+961%' AND LEN(WhishNumber) BETWEEN 11 AND 12 AND SUBSTRING(WhishNumber,2,20) NOT LIKE '%[^0-9]%'),
 Status varchar(12) NOT NULL CONSTRAINT DF_Withdrawal_Status DEFAULT 'Requested' CONSTRAINT CK_Withdrawal_Status CHECK(Status IN('Requested','Sent','Rejected')),
 SentReference nvarchar(100) NULL, ReviewedBy int NULL REFERENCES dbo.Users(UserId), ReviewNote nvarchar(400) NULL,
 CreatedAt datetime2(3) NOT NULL CONSTRAINT DF_Withdrawal_Created DEFAULT SYSUTCDATETIME(), ReviewedAt datetime2(3) NULL,
 CONSTRAINT UQ_Withdrawal_Key UNIQUE(UserId,IdempotencyKey));
CREATE UNIQUE INDEX UX_Withdrawal_Reference ON dbo.WalletWithdrawals(SentReference) WHERE SentReference IS NOT NULL;
CREATE TABLE dbo.WalletLedger(EntryId bigint IDENTITY CONSTRAINT PK_WalletLedger PRIMARY KEY, UserId int NOT NULL REFERENCES dbo.Users(UserId),
 Amount decimal(12,2) NOT NULL CONSTRAINT CK_Ledger_Amount CHECK(Amount<>0), BalanceAfter decimal(12,2) NOT NULL CONSTRAINT CK_Ledger_Balance CHECK(BalanceAfter>=0),
 Kind varchar(20) NOT NULL CONSTRAINT CK_Ledger_Kind CHECK(Kind IN('TopUp','DepositPayment','DepositRefund','Withdrawal','WithdrawalReversal')),
 TopUpId uniqueidentifier NULL REFERENCES dbo.WalletTopUps(TopUpId), WithdrawalId uniqueidentifier NULL REFERENCES dbo.WalletWithdrawals(WithdrawalId),
 DepositID uniqueidentifier NULL REFERENCES dbo.DepositBookings(DepositID), Note nvarchar(400) NULL, CreatedBy int NULL REFERENCES dbo.Users(UserId),
 CreatedAt datetime2(3) NOT NULL CONSTRAINT DF_Ledger_Created DEFAULT SYSUTCDATETIME(),
 CONSTRAINT CK_Ledger_Link CHECK((Kind='TopUp' AND TopUpId IS NOT NULL AND Amount>0) OR (Kind='DepositPayment' AND DepositID IS NOT NULL AND Amount<0)
  OR (Kind='DepositRefund' AND DepositID IS NOT NULL AND Amount>0) OR (Kind='Withdrawal' AND WithdrawalId IS NOT NULL AND Amount<0)
  OR (Kind='WithdrawalReversal' AND WithdrawalId IS NOT NULL AND Amount>0)));
-- Each money event can be posted exactly once.
CREATE UNIQUE INDEX UX_Ledger_TopUp ON dbo.WalletLedger(TopUpId) WHERE TopUpId IS NOT NULL;
CREATE UNIQUE INDEX UX_Ledger_Deposit ON dbo.WalletLedger(Kind,DepositID) WHERE DepositID IS NOT NULL;
CREATE UNIQUE INDEX UX_Ledger_Withdrawal ON dbo.WalletLedger(Kind,WithdrawalId) WHERE WithdrawalId IS NOT NULL;
CREATE INDEX IX_Ledger_User ON dbo.WalletLedger(UserId,EntryId DESC) INCLUDE(Amount,BalanceAfter,Kind,CreatedAt);
-- ===== Deposits can be paid from the wallet: $1 now, venue share + fee in cash at the venue =====
ALTER TABLE dbo.DepositBookings DROP CONSTRAINT CK__DepositBo__Payme__2057CCD0;
ALTER TABLE dbo.DepositBookings ADD CONSTRAINT CK_Deposit_Method CHECK(PaymentMethod IN('WhishManual','Cash','Wallet'));
ALTER TABLE dbo.UserActivityLogs DROP CONSTRAINT CK_UserActivity_Action;
ALTER TABLE dbo.UserActivityLogs ADD CONSTRAINT CK_UserActivity_Action CHECK(Action IN('DiscoverySearch','EventViewed','FilterApplied','JoinAttempt','PaymentOptionsViewed','DepositInitialized','DepositSubmitted','DepositReviewed','DepositWallet'));
ALTER TABLE dbo.DepositBookings DROP COLUMN DueNow;
ALTER TABLE dbo.DepositBookings ADD DueNow AS (CASE WHEN PaymentMethod IN('Cash','Wallet') THEN DepositAmount ELSE BaseAmount+ServiceFee+DepositAmount END);
GO
CREATE OR ALTER PROCEDURE dbo.WalletPost @UserId int,@Amount decimal(12,2),@Kind varchar(20),@TopUpId uniqueidentifier=NULL,@WithdrawalId uniqueidentifier=NULL,
 @DepositID uniqueidentifier=NULL,@Note nvarchar(400)=NULL,@By int=NULL AS
BEGIN
 SET NOCOUNT ON;
 -- Caller owns the transaction. The account row lock serializes every balance change for one user.
 IF NOT EXISTS(SELECT 1 FROM dbo.WalletAccounts WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId) INSERT dbo.WalletAccounts(UserId) VALUES(@UserId);
 DECLARE @Balance decimal(12,2);
 SELECT @Balance=Balance FROM dbo.WalletAccounts WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId;
 IF @Balance+@Amount<0 THROW 50006,'Not enough money in your GameOn wallet. Top up first.',1;
 IF @Balance+@Amount>500 THROW 50002,'A GameOn wallet can hold at most USD 500.',1;
 UPDATE dbo.WalletAccounts SET Balance=Balance+@Amount,UpdatedAt=SYSUTCDATETIME() WHERE UserId=@UserId;
 INSERT dbo.WalletLedger(UserId,Amount,BalanceAfter,Kind,TopUpId,WithdrawalId,DepositID,Note,CreatedBy)
 VALUES(@UserId,@Amount,@Balance+@Amount,@Kind,@TopUpId,@WithdrawalId,@DepositID,@Note,@By);
END;
GO
CREATE OR ALTER TRIGGER dbo.WalletDepositRefund ON dbo.DepositBookings AFTER UPDATE AS
BEGIN
 SET NOCOUNT ON;
 IF NOT UPDATE(DepositStatus) RETURN;
 -- A wallet-paid deposit that becomes refundable is credited back inside GameOn immediately.
 -- This is real settlement: the money never left the GameOn ledger.
 DECLARE @D uniqueidentifier,@U int,@A decimal(12,2),@G int;
 DECLARE r CURSOR LOCAL FAST_FORWARD FOR
  SELECT i.DepositID,i.UserId,i.DepositAmount,i.GameId FROM inserted i JOIN deleted d ON d.DepositID=i.DepositID
  WHERE i.PaymentMethod='Wallet' AND i.DepositStatus IN('REFUND_PENDING','REFUNDED') AND d.DepositStatus NOT IN('REFUND_PENDING','REFUNDED')
   AND NOT EXISTS(SELECT 1 FROM dbo.WalletLedger l WHERE l.Kind='DepositRefund' AND l.DepositID=i.DepositID);
 OPEN r; FETCH NEXT FROM r INTO @D,@U,@A,@G;
 WHILE @@FETCH_STATUS=0
 BEGIN
  EXEC dbo.WalletPost @UserId=@U,@Amount=@A,@Kind='DepositRefund',@DepositID=@D,@Note=N'$1 commitment deposit returned to wallet';
  UPDATE dbo.DepositBookings SET DepositStatus='REFUNDED',RefundReference=N'WALLET-'+CONVERT(nvarchar(36),@D),RefundedAt=SYSUTCDATETIME() WHERE DepositID=@D;
  UPDATE dbo.Participations SET DepositStatus='REFUNDED' WHERE GameId=@G AND UserId=@U;
  INSERT dbo.Notifications(UserId,GameId,Title,Body) VALUES(@U,@G,N'$1 is back in your wallet',N'Your commitment deposit was returned to your GameOn wallet. Use it for your next game or withdraw it to Whish.');
  FETCH NEXT FROM r INTO @D,@U,@A,@G;
 END;
 CLOSE r; DEALLOCATE r;
END;
GO
CREATE OR ALTER PROCEDURE dbo.JoinWithWallet @GameId int,@UserId int,@Key uniqueidentifier,@Revision int,@ClaimSponsored bit=0 AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON; SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
 IF @Key IS NULL THROW 50002,'Missing request key.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Capacity int,@Cost decimal(12,2),@State varchar(12),@Start datetime2(3),@Current int,@Required bit,@Currency char(3),@Host int,@Slots int,@ID uniqueidentifier;
  SELECT @Capacity=Capacity,@Cost=CostPerPlayer,@State=GameStatus,@Start=StartsAt,@Current=Revision,@Required=RequiresDeposit,@Currency=Currency,@Host=HostUserId,@Slots=SponsoredSlots
  FROM dbo.GamesAndEvents WITH(UPDLOCK,HOLDLOCK) WHERE EventId=@GameId;
  IF @Capacity IS NULL THROW 50001,'Game not found.',1;
  -- Same request again (double tap, retry): return the original result, never charge twice.
  SELECT @ID=DepositID FROM dbo.DepositBookings WHERE UserId=@UserId AND IdempotencyKey=@Key;
  IF @ID IS NOT NULL
  BEGIN
   IF NOT EXISTS(SELECT 1 FROM dbo.DepositBookings WHERE DepositID=@ID AND GameId=@GameId AND PaymentMethod='Wallet') THROW 50004,'Request key conflicts with another booking.',1;
   COMMIT; SELECT 'Joined' AS Status,@ID AS DepositId,(SELECT Balance FROM dbo.WalletAccounts WHERE UserId=@UserId) AS WalletBalance; RETURN;
  END;
  IF @UserId=@Host THROW 50002,'You are the host of this game.',1;
  IF @State<>'Open' OR @Start<=SYSUTCDATETIME() THROW 50002,'This game is no longer open.',1;
  IF @Required<>1 OR @Currency<>'USD' THROW 50002,'This game does not take wallet deposits.',1;
  IF @Revision<>@Current THROW 50004,'The host changed this game. Refresh to see the latest details.',1;
  IF EXISTS(SELECT 1 FROM dbo.Users WHERE UserId=@Host AND HostHoldAt IS NOT NULL) THROW 50004,'This host is under review. GameOn is not taking deposits for their games.',1;
  IF EXISTS(SELECT 1 FROM dbo.Participations WHERE GameId=@GameId AND UserId=@UserId) THROW 50004,'You already have a place in this game.',1;
  IF EXISTS(SELECT 1 FROM dbo.DepositBookings WHERE GameId=@GameId AND UserId=@UserId AND DepositStatus IN('PENDING','PAID_HELD','DISPUTED')) THROW 50004,'You already have a deposit for this game. Check Payments.',1;
  IF (SELECT COUNT(*) FROM dbo.Participations WHERE GameId=@GameId)>=@Capacity THROW 50004,'This game is full. Join the waitlist instead (no payment needed).',1;
  IF @ClaimSponsored=1 AND @Slots<=(SELECT COUNT(*) FROM dbo.SponsoredSeats WHERE GameId=@GameId) THROW 50004,'No host-covered places remain.',1;
  IF @ClaimSponsored=1 SET @Cost=0;
  SET @ID=NEWID();
  INSERT dbo.DepositBookings(DepositID,GameId,UserId,IdempotencyKey,EventRevision,BaseAmount,ServiceFee,ClaimSponsored,RecipientNumber,PaymentMethod)
  VALUES(@ID,@GameId,@UserId,@Key,@Current,@Cost,dbo.PlatformServiceFee(@Cost),@ClaimSponsored,'+96171364432','Wallet');
  EXEC dbo.WalletPost @UserId=@UserId,@Amount=-1.00,@Kind='DepositPayment',@DepositID=@ID,@Note=N'$1 commitment deposit';
  UPDATE dbo.DepositBookings SET DepositStatus='PAID_HELD',DepositPaidAt=GETUTCDATE(),ReviewNote=N'Paid instantly from GameOn wallet.' WHERE DepositID=@ID;
  DELETE dbo.Waitlists WHERE GameId=@GameId AND UserId=@UserId;
  DECLARE @JoinResult varchar(20);
  EXEC dbo.sp_JoinGameOrActivity @GameId=@GameId,@UserId=@UserId,@ClaimSponsored=@ClaimSponsored,@PaymentMethod='Cash',@CashIdempotencyKey=@ID,
   @ResultStatus=@JoinResult OUTPUT,@ReturnResult=0,@DepositAuthorizationID=@ID;
  IF @JoinResult<>'Joined' THROW 50004,'Your place could not be confirmed. Nothing was charged.',1;
  INSERT dbo.UserActivityLogs(UserId,EventId,Action,DepositStatus,DepositAmount,RequiresDeposit) VALUES(@UserId,@GameId,'DepositWallet','PAID_HELD',1,1);
  COMMIT;
  SELECT 'Joined' AS Status,@ID AS DepositId,(SELECT Balance FROM dbo.WalletAccounts WHERE UserId=@UserId) AS WalletBalance;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.GetWallet @UserId int AS
BEGIN
 SET NOCOUNT ON;
 SELECT COALESCE((SELECT Balance FROM dbo.WalletAccounts WHERE UserId=@UserId),0) AS Balance,
  COALESCE((SELECT SUM(Amount) FROM dbo.WalletWithdrawals WHERE UserId=@UserId AND Status='Requested'),0) AS PendingWithdrawals,
  COALESCE((SELECT SUM(Amount) FROM dbo.WalletTopUps WHERE UserId=@UserId AND Status='Submitted'),0) AS PendingTopUps,
  CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@UserId) THEN 1 ELSE 0 END AS bit) AS CanReview;
 SELECT TOP(50) l.EntryId,l.Amount,l.BalanceAfter,l.Kind,l.Note,l.CreatedAt,d.GameId FROM dbo.WalletLedger l LEFT JOIN dbo.DepositBookings d ON d.DepositID=l.DepositID
 WHERE l.UserId=@UserId ORDER BY l.EntryId DESC;
 SELECT TOP(20) TopUpId,Amount,Currency,RecipientNumber,TransferReference,Status,ReviewNote,CreatedAt FROM dbo.WalletTopUps WHERE UserId=@UserId AND Status IN('AwaitingReference','Submitted','Rejected') ORDER BY CreatedAt DESC;
 SELECT TOP(20) WithdrawalId,Amount,WhishNumber,Status,SentReference,ReviewNote,CreatedAt FROM dbo.WalletWithdrawals WHERE UserId=@UserId ORDER BY CreatedAt DESC;
END;
GO
CREATE OR ALTER PROCEDURE dbo.CreateTopUp @UserId int,@Key uniqueidentifier,@Amount decimal(12,2),@Recipient varchar(20) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF @Amount IS NULL OR @Amount NOT BETWEEN 1 AND 200 OR @Amount<>ROUND(@Amount,2) THROW 50002,'Top up between USD 1 and 200.',1;
 IF @Recipient<>'+96171364432' THROW 50002,'Top-ups are unavailable.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @ID uniqueidentifier=(SELECT TopUpId FROM dbo.WalletTopUps WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId AND IdempotencyKey=@Key);
  IF @ID IS NULL
  BEGIN
   IF (SELECT COUNT(*) FROM dbo.WalletTopUps WHERE UserId=@UserId AND Status IN('AwaitingReference','Submitted'))>=3 THROW 50004,'You already have 3 top-ups waiting. Finish or cancel one first.',1;
   SET @ID=NEWID();
   INSERT dbo.WalletTopUps(TopUpId,UserId,IdempotencyKey,Amount,RecipientNumber) VALUES(@ID,@UserId,@Key,@Amount,@Recipient);
  END;
  COMMIT;
  SELECT TopUpId,Amount,Currency,RecipientNumber,TransferReference,Status FROM dbo.WalletTopUps WHERE TopUpId=@ID;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.SubmitTopUpReference @UserId int,@TopUpId uniqueidentifier,@Reference nvarchar(100) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 SET @Reference=TRIM(@Reference);
 IF @Reference IS NULL OR LEN(@Reference) NOT BETWEEN 4 AND 100 THROW 50002,'Enter the Whish transaction reference (4-100 characters).',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Status varchar(20),@Old nvarchar(100);
  SELECT @Status=Status,@Old=TransferReference FROM dbo.WalletTopUps WITH(UPDLOCK,HOLDLOCK) WHERE TopUpId=@TopUpId AND UserId=@UserId;
  IF @Status IS NULL THROW 50001,'Top-up not found.',1;
  IF @Status='Submitted' AND @Old=@Reference BEGIN COMMIT; SELECT TopUpId,Amount,Status,TransferReference FROM dbo.WalletTopUps WHERE TopUpId=@TopUpId; RETURN; END;
  IF @Status<>'AwaitingReference' THROW 50004,'This top-up was already submitted or reviewed.',1;
  IF EXISTS(SELECT 1 FROM dbo.WalletTopUps WHERE TransferReference=@Reference) OR EXISTS(SELECT 1 FROM dbo.VerifiedWhishTransfers WHERE TransferReference=@Reference)
   OR EXISTS(SELECT 1 FROM dbo.DepositBookings WHERE TransferReference=@Reference) THROW 50004,'This Whish reference was already used.',1;
  UPDATE dbo.WalletTopUps SET TransferReference=@Reference,Status='Submitted' WHERE TopUpId=@TopUpId;
  COMMIT;
  SELECT TopUpId,Amount,Status,TransferReference FROM dbo.WalletTopUps WHERE TopUpId=@TopUpId;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.CancelTopUp @UserId int,@TopUpId uniqueidentifier AS
BEGIN
 SET NOCOUNT ON;
 UPDATE dbo.WalletTopUps SET Status='Cancelled' WHERE TopUpId=@TopUpId AND UserId=@UserId AND Status='AwaitingReference';
 IF @@ROWCOUNT=0 THROW 50004,'Only a top-up without a submitted reference can be cancelled.',1;
 SELECT 'Cancelled' AS Status;
END;
GO
CREATE OR ALTER PROCEDURE dbo.ReviewTopUp @OperatorId int,@TopUpId uniqueidentifier,@Decision varchar(8),@Note nvarchar(400) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF NOT EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@OperatorId) THROW 50003,'Payment operator access required.',1;
 IF @Decision NOT IN('VERIFY','REJECT') OR @Note IS NULL OR LEN(TRIM(@Note))<4 THROW 50002,'Choose a decision and write what you checked.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Owner int,@Status varchar(20),@Amount decimal(12,2);
  SELECT @Owner=UserId,@Status=Status,@Amount=Amount FROM dbo.WalletTopUps WITH(UPDLOCK,HOLDLOCK) WHERE TopUpId=@TopUpId;
  IF @Owner IS NULL THROW 50001,'Top-up not found.',1;
  IF @Owner=@OperatorId THROW 50003,'Operators cannot review their own top-ups.',1;
  IF @Status<>'Submitted' THROW 50004,'This top-up is not waiting for review.',1;
  UPDATE dbo.WalletTopUps SET Status=CASE WHEN @Decision='VERIFY' THEN 'Verified' ELSE 'Rejected' END,ReviewedBy=@OperatorId,ReviewNote=TRIM(@Note),ReviewedAt=SYSUTCDATETIME() WHERE TopUpId=@TopUpId;
  IF @Decision='VERIFY'
  BEGIN
   EXEC dbo.WalletPost @UserId=@Owner,@Amount=@Amount,@Kind='TopUp',@TopUpId=@TopUpId,@Note=N'Whish top-up verified',@By=@OperatorId;
   INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,N'Wallet topped up',N'USD '+CONVERT(nvarchar(20),@Amount)+N' is now in your GameOn wallet.');
  END
  ELSE INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,N'Top-up not confirmed',LEFT(N'We could not find this Whish transfer. '+TRIM(@Note),800));
  COMMIT;
  SELECT CASE WHEN @Decision='VERIFY' THEN 'Verified' ELSE 'Rejected' END AS Status;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.RequestWithdrawal @UserId int,@Key uniqueidentifier,@Amount decimal(12,2),@WhishNumber varchar(20) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF @Amount IS NULL OR @Amount NOT BETWEEN 1 AND 500 OR @Amount<>ROUND(@Amount,2) THROW 50002,'Withdraw between USD 1 and your balance.',1;
 IF @WhishNumber IS NULL OR @WhishNumber NOT LIKE '+961%' OR LEN(@WhishNumber) NOT BETWEEN 11 AND 12 OR SUBSTRING(@WhishNumber,2,20) LIKE '%[^0-9]%' THROW 50002,'Enter your Lebanese Whish number, e.g. +96171123456.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @ID uniqueidentifier=(SELECT WithdrawalId FROM dbo.WalletWithdrawals WITH(UPDLOCK,HOLDLOCK) WHERE UserId=@UserId AND IdempotencyKey=@Key);
  IF @ID IS NULL
  BEGIN
   SET @ID=NEWID();
   INSERT dbo.WalletWithdrawals(WithdrawalId,UserId,IdempotencyKey,Amount,WhishNumber) VALUES(@ID,@UserId,@Key,@Amount,@WhishNumber);
   DECLARE @Neg decimal(12,2)=-@Amount;
   EXEC dbo.WalletPost @UserId=@UserId,@Amount=@Neg,@Kind='Withdrawal',@WithdrawalId=@ID,@Note=N'Withdrawal to Whish requested';
  END;
  COMMIT;
  SELECT WithdrawalId,Amount,WhishNumber,Status FROM dbo.WalletWithdrawals WHERE WithdrawalId=@ID;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.ReviewWithdrawal @OperatorId int,@WithdrawalId uniqueidentifier,@Decision varchar(8),@Reference nvarchar(100)=NULL,@Note nvarchar(400) AS
BEGIN
 SET NOCOUNT ON; SET XACT_ABORT ON;
 IF NOT EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@OperatorId) THROW 50003,'Payment operator access required.',1;
 IF @Decision NOT IN('SENT','REJECT') OR @Note IS NULL OR LEN(TRIM(@Note))<4 THROW 50002,'Choose a decision and write a note.',1;
 IF @Decision='SENT' AND (@Reference IS NULL OR LEN(TRIM(@Reference)) NOT BETWEEN 4 AND 100) THROW 50002,'Record the Whish reference of the money you sent.',1;
 BEGIN TRY
  BEGIN TRANSACTION;
  DECLARE @Owner int,@Status varchar(12),@Amount decimal(12,2);
  SELECT @Owner=UserId,@Status=Status,@Amount=Amount FROM dbo.WalletWithdrawals WITH(UPDLOCK,HOLDLOCK) WHERE WithdrawalId=@WithdrawalId;
  IF @Owner IS NULL THROW 50001,'Withdrawal not found.',1;
  IF @Owner=@OperatorId THROW 50003,'Operators cannot review their own withdrawals.',1;
  IF @Status<>'Requested' THROW 50004,'This withdrawal was already handled.',1;
  IF @Decision='SENT'
  BEGIN
   IF EXISTS(SELECT 1 FROM dbo.WalletWithdrawals WHERE SentReference=TRIM(@Reference)) THROW 50004,'This Whish reference is already recorded.',1;
   UPDATE dbo.WalletWithdrawals SET Status='Sent',SentReference=TRIM(@Reference),ReviewedBy=@OperatorId,ReviewNote=TRIM(@Note),ReviewedAt=SYSUTCDATETIME() WHERE WithdrawalId=@WithdrawalId;
   INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,N'Withdrawal sent',N'USD '+CONVERT(nvarchar(20),@Amount)+N' was sent to your Whish. Reference '+TRIM(@Reference)+N'.');
  END
  ELSE
  BEGIN
   UPDATE dbo.WalletWithdrawals SET Status='Rejected',ReviewedBy=@OperatorId,ReviewNote=TRIM(@Note),ReviewedAt=SYSUTCDATETIME() WHERE WithdrawalId=@WithdrawalId;
   EXEC dbo.WalletPost @UserId=@Owner,@Amount=@Amount,@Kind='WithdrawalReversal',@WithdrawalId=@WithdrawalId,@Note=N'Withdrawal not sent; money returned to wallet',@By=@OperatorId;
   INSERT dbo.Notifications(UserId,Title,Body) VALUES(@Owner,N'Withdrawal returned to wallet',LEFT(TRIM(@Note),800));
  END;
  COMMIT;
  SELECT CASE WHEN @Decision='SENT' THEN 'Sent' ELSE 'Rejected' END AS Status;
 END TRY BEGIN CATCH IF XACT_STATE()<>0 ROLLBACK; THROW; END CATCH;
END;
GO
CREATE OR ALTER PROCEDURE dbo.ListWalletReviews @OperatorId int AS
BEGIN
 SET NOCOUNT ON;
 IF NOT EXISTS(SELECT 1 FROM dbo.PaymentOperators WHERE UserId=@OperatorId) THROW 50003,'Payment operator access required.',1;
 SELECT TOP(200) t.TopUpId,t.Amount,t.Currency,t.RecipientNumber,t.TransferReference,t.CreatedAt,u.DisplayName,t.UserId FROM dbo.WalletTopUps t JOIN dbo.Users u ON u.UserId=t.UserId
 WHERE t.Status='Submitted' ORDER BY t.CreatedAt;
 SELECT TOP(200) w.WithdrawalId,w.Amount,w.WhishNumber,w.CreatedAt,u.DisplayName,w.UserId FROM dbo.WalletWithdrawals w JOIN dbo.Users u ON u.UserId=w.UserId
 WHERE w.Status='Requested' ORDER BY w.CreatedAt;
 -- Reconciliation: the wallet balances GameOn owes users must equal verified top-ups minus what left.
 SELECT (SELECT COALESCE(SUM(Balance),0) FROM dbo.WalletAccounts) AS TotalHeldForUsers,
  (SELECT COALESCE(SUM(Amount),0) FROM dbo.WalletLedger) AS LedgerSum,
  (SELECT COALESCE(SUM(Amount),0) FROM dbo.WalletWithdrawals WHERE Status='Requested') AS AwaitingPayout;
END;
